getpoints = function(ob) {
    type = ob.type; data = ob.data;
    switch(type) {
        case: 'lineseg': return(data);
        case: 'circle': return([data[0]]); // returns array with only 1 point
    }
}

move = function(ob, moveVec) {
    var t = getpoints(ob);
    for (i=0; i<t.length; i++) {
        t[i][0]+=moveVec[0];
        t[i][1]+=moveVec[1];
    }
}

scale = function(ob, base, scale) {
    var ps = getpoints(ob);
    var ls = getlength(ob);
    for (i=0; i<ps.length; i++) {
        d[0] = ps[i][0] - base[0]
        ps[i][0] = base[0] + d[0]*scale;
    }
    for (i=0; i<ls.length; i++) ls[i]*=scale;
}
